4use core::sync::atomic::{AtomicUsize, Ordering};
5
6/// The type of an event handler.
7///
8/// Currently no arguments and return values are supported.
9pub type Handler = fn();
10
11/// A lock-free table of event handlers.
12///
13/// It internally uses an array of `AtomicUsize` to store the handlers.
14pub struct HandlerTable<const N: usize> {
15    handlers: [AtomicUsize; N],
16}
17
18impl<const N: usize> HandlerTable<N> {
19    /// Creates a new handler table with all entries empty.
20    pub const fn new() -> Self {
21        Self {
22            handlers: [const { AtomicUsize::new(0) }; N],
23        }
24    }
25
26    /// Registers a handler for the given index.
27    ///
28    /// Returns `true` if the registration succeeds, `false` if the index is out
29    /// of bounds or the handler is already registered.
30    pub fn register_handler(&self, idx: usize, handler: Handler) -> bool {
31        if idx >= N {
32            return false;
33        }
34        self.handlers[idx]
35            .compare_exchange(0, handler as usize, Ordering::Acquire, Ordering::Relaxed)
36            .is_ok()
37    }
38
39    /// Unregisters the handler for the given index.
40    ///
41    /// Returns the existing handler if it is registered, `None` otherwise.
42    pub fn unregister_handler(&self, idx: usize) -> Option<Handler> {
43        if idx >= N {
44            return None;
45        }
46        let handler = self.handlers[idx].swap(0, Ordering::Acquire);
47        if handler != 0 {
48            Some(unsafe { core::mem::transmute::<usize, fn()>(handler) })
49        } else {
50            None
51        }
52    }
53
54    /// Handles the event with the given index.
55    ///
56    /// Returns `true` if the event is handled, `false` if no handler is
57    /// registered for the given index.
58    pub fn handle(&self, idx: usize) -> bool {
59        if idx >= N {
60            return false;
61        }
62        let handler = self.handlers[idx].load(Ordering::Acquire);
63        if handler != 0 {
64            let handler: Handler = unsafe { core::mem::transmute(handler) };
65            handler();
66            true
67        } else {
68            false
69        }
70    }
71}
72
73impl<const N: usize> Default for HandlerTable<N> {
74    fn default() -> Self {
75        Self::new()
76    }
77}
